cmake_minimum_required(VERSION 3.16)
project(FieldInspectionInference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenCV REQUIRED)
set(SDK /userdata/ai-workbench/vendor/rknn-llm)
set(VISION ${SDK}/examples/multimodal_model_demo/deploy)
set(RKLLM ${SDK}/rkllm-runtime/Linux/librkllm_api)
set(RKNN ${VISION}/3rdparty/librknnrt/Linux/librknn_api)
add_executable(native_runner native_runner.cpp ${VISION}/src/image_enc.cc)
target_include_directories(native_runner PRIVATE ${RKLLM}/include ${RKNN}/include ${VISION}/src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(native_runner PRIVATE Qt6::Core ${OpenCV_LIBS} ${RKNN}/aarch64/librknnrt.so ${RKLLM}/aarch64/librkllmrt.so)
target_link_options(native_runner PRIVATE -Wl,--allow-shlib-undefined -Wl,-Bsymbolic)
set_target_properties(native_runner PROPERTIES BUILD_RPATH "${RKNN}/aarch64;${RKLLM}/aarch64")
